cmake_minimum_required(VERSION 3.22)
project(jhbase VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.build" jb_build_root)
file(REAL_PATH "${CMAKE_BINARY_DIR}" jb_build_dir)
string(FIND "${jb_build_dir}/" "${jb_build_root}/" jb_build_prefix)
if(NOT jb_build_prefix EQUAL 0 OR jb_build_dir STREQUAL jb_build_root)
  message(FATAL_ERROR "Use a build directory below jhbase/.build/")
endif()

set(JHBASE_HAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../libraries/JaszczurHAL"
    CACHE PATH "HAL supplying only the prototype's status and array headers")
set(JHBASE_UNITY_ROOT "${JHBASE_HAL_ROOT}/third_party/Unity"
    CACHE PATH "Existing pinned Unity checkout; nothing is downloaded")
option(JHBASE_SANITIZERS "Instrument library and tests with address/undefined sanitizers" OFF)
option(JHBASE_FUZZ "Build the Clang libFuzzer entrypoint" OFF)
option(JHBASE_LTO "Enable interprocedural optimization" OFF)
include(CTest)

include(cmake/dependencies.cmake)
add_library(jb_status INTERFACE)
add_library(jb::status ALIAS jb_status)
target_include_directories(jb_status INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include" "${JHBASE_HAL_ROOT}/src")
if(NOT MSVC)
  target_compile_options(jb_status INTERFACE -fno-short-enums)
endif()

add_library(jhbase STATIC src/utf8.c src/json_parse.c src/json_string.c src/json_read.c)
add_library(jb::base ALIAS jhbase)
target_link_libraries(jhbase PUBLIC jb_status)
target_compile_features(jhbase PUBLIC c_std_11)
set_target_properties(jhbase PROPERTIES C_EXTENSIONS OFF)

function(jb_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic
      -Wconversion -Wshadow -Wstrict-prototypes -fstrict-aliasing)
  endif()
endfunction()
jb_warnings(jhbase)

if(JHBASE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT jb_ipo OUTPUT jb_ipo_error)
  if(NOT jb_ipo)
    message(FATAL_ERROR "LTO unavailable: ${jb_ipo_error}")
  endif()
  set_property(TARGET jhbase PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
if(JHBASE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use the native MSVC tests; this sanitizer profile needs GCC/Clang")
  endif()
  target_compile_options(jhbase PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jhbase PUBLIC -fsanitize=address,undefined)
endif()
if(JHBASE_FUZZ)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "JHBASE_FUZZ requires Clang with libFuzzer")
  endif()
  target_compile_options(jhbase PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(fuzz_json tests/fuzz_json.c)
  target_link_libraries(fuzz_json PRIVATE jb::base)
  target_compile_options(fuzz_json PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(fuzz_json PRIVATE -fsanitize=fuzzer,address,undefined)
endif()

if(BUILD_TESTING)
  jb_check_unity()
  add_library(unity STATIC "${JHBASE_UNITY_ROOT}/src/unity.c")
  target_include_directories(unity PUBLIC "${JHBASE_UNITY_ROOT}/src")
  target_compile_definitions(unity PUBLIC UNITY_INCLUDE_DOUBLE)
  foreach(name utf8 json)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE jb::base unity)
    jb_warnings(test_${name})
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 60 LABELS "basic")
  endforeach()
  add_executable(test_cxx tests/test_cxx.cpp)
  target_link_libraries(test_cxx PRIVATE jb::base)
  target_compile_features(test_cxx PRIVATE cxx_std_17)
  set_target_properties(test_cxx PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(test_cxx PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(test_cxx PRIVATE -Wall -Wextra -Werror -Wpedantic)
  endif()
  add_test(NAME cxx COMMAND test_cxx)
  set_tests_properties(cxx PROPERTIES TIMEOUT 30 LABELS "basic")
  add_test(NAME abi COMMAND "${CMAKE_COMMAND}"
    "-DJB_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
    "-DJB_HAL=${JHBASE_HAL_ROOT}"
    "-DJB_WORK=${CMAKE_CURRENT_BINARY_DIR}/abi"
    "-DJB_CC=${CMAKE_C_COMPILER}" "-DJB_CXX=${CMAKE_CXX_COMPILER}"
    "-DJB_MSVC=${MSVC}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_abi.cmake")
  set_tests_properties(abi PROPERTIES TIMEOUT 60 LABELS "basic")
  if(JHBASE_LTO)
    set_property(TARGET test_utf8 test_json test_cxx
                 PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
