cmake_minimum_required(VERSION 3.22)
project(jhbase VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# A standalone build stays below jhbase/.build/ and owns the tests. A parent
# project (HAL) keeps its own build tree and takes only the library targets.
if(PROJECT_IS_TOP_LEVEL)
  file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.build" jb_build_root)
  file(REAL_PATH "${CMAKE_BINARY_DIR}" jb_build_dir)
  string(FIND "${jb_build_dir}/" "${jb_build_root}/" jb_build_prefix)
  if(NOT jb_build_prefix EQUAL 0 OR jb_build_dir STREQUAL jb_build_root)
    message(FATAL_ERROR "Use a build directory below jhbase/.build/")
  endif()
  set(JHBASE_HAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../JaszczurHAL"
      CACHE PATH "HAL checkout supplying the numeric test fixture")
  set(JHBASE_TOOLS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.build/tools"
      CACHE PATH "Pinned gate tools placed there by scripts/ensure_tools.sh")
  set(JHBASE_UNITY_ROOT "${JHBASE_TOOLS_ROOT}/Unity"
      CACHE PATH "Unity checkout pinned in third_party/unity_version.conf; nothing is downloaded")
  include(CTest)
endif()
option(JHBASE_SANITIZERS "Instrument library and tests with address/undefined sanitizers" OFF)
option(JHBASE_FUZZ "Build the Clang libFuzzer entrypoint" OFF)
option(JHBASE_LTO "Enable interprocedural optimization" OFF)
option(JHBASE_COMPARE "Build explicit pinned upstream parser comparison" OFF)
option(JHBASE_VSCODE_CHECK "Register the VS Code workspace check as a CTest test" ON)

include(GNUInstallDirs)
include(cmake/dependencies.cmake)
add_library(jb_status INTERFACE)
add_library(jb::status ALIAS jb_status)
set_target_properties(jb_status PROPERTIES EXPORT_NAME status)
target_include_directories(jb_status INTERFACE
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")

add_library(jb_ryu OBJECT third_party/ryu/ryu/f2s.c third_party/ryu/ryu/d2s.c)
target_include_directories(jb_ryu PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ryu")
target_compile_definitions(jb_ryu PRIVATE RYU_OPTIMIZE_SIZE RYU_ONLY_64_BIT_OPS NDEBUG)
add_library(jhbase STATIC src/utf8.c src/json_parse.c src/json_string.c src/json_read.c
  src/number_big.c src/json_number.c src/number_format.c src/json_writer.c src/literal.c)
target_sources(jhbase PRIVATE $<TARGET_OBJECTS:jb_ryu>)
target_include_directories(jhbase PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ryu")
add_library(jb::base ALIAS jhbase)
set_target_properties(jhbase PROPERTIES EXPORT_NAME base)
# In a parent project the library builds only when something links it; the
# install rules stay active, which EXCLUDE_FROM_ALL on add_subdirectory drops.
if(NOT PROJECT_IS_TOP_LEVEL)
  set_target_properties(jhbase jb_ryu PROPERTIES EXCLUDE_FROM_ALL TRUE)
endif()
target_link_libraries(jhbase PUBLIC jb_status)
target_compile_features(jhbase PUBLIC c_std_11)
set_target_properties(jhbase PROPERTIES C_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(jhbase PRIVATE /Gy /Gw)
  target_compile_options(jb_ryu PRIVATE /Gy /Gw)
  target_link_options(jhbase INTERFACE /OPT:REF)
else()
  target_compile_options(jhbase PRIVATE -ffunction-sections -fdata-sections)
  target_compile_options(jb_ryu PRIVATE -ffunction-sections -fdata-sections)
  target_link_options(jhbase INTERFACE -Wl,--gc-sections)
endif()

function(jb_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wpedantic
      -Wconversion -Wshadow -Wvla $<$<COMPILE_LANGUAGE:C>:-Wstrict-prototypes> -fstrict-aliasing)
  endif()
endfunction()
jb_warnings(jhbase)
include(cmake/install.cmake)

if(JHBASE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT jb_ipo OUTPUT jb_ipo_error)
  if(NOT jb_ipo)
    message(FATAL_ERROR "LTO unavailable: ${jb_ipo_error}")
  endif()
  set_property(TARGET jhbase PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()

if(JHBASE_COMPARE)
  include(cmake/comparison.cmake)
endif()
if(JHBASE_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "Use the native MSVC tests; this sanitizer profile needs GCC/Clang")
  endif()
  target_compile_options(jhbase PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(jb_ryu PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jhbase PUBLIC -fsanitize=address,undefined)
endif()
if(JHBASE_FUZZ)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "JHBASE_FUZZ requires Clang with libFuzzer")
  endif()
  target_compile_options(jhbase PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_compile_options(jb_ryu PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  foreach(name json numbers writer)
    add_executable(fuzz_${name} tests/fuzz_${name}.c)
    target_link_libraries(fuzz_${name} PRIVATE jb::base)
    target_compile_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    target_link_options(fuzz_${name} PRIVATE -fsanitize=fuzzer,address,undefined)
    jb_warnings(fuzz_${name})
  endforeach()
endif()

if(PROJECT_IS_TOP_LEVEL AND BUILD_TESTING)
  jb_check_unity()
  add_library(unity STATIC "${JHBASE_UNITY_ROOT}/src/unity.c")
  target_include_directories(unity PUBLIC "${JHBASE_UNITY_ROOT}/src")
  # The tests assert 64-bit values; Unity enables them by itself only where long
  # or pointers are 64 bits wide, which excludes MSVC Win32 and other ILP32 hosts.
  target_compile_definitions(unity PUBLIC UNITY_INCLUDE_DOUBLE UNITY_SUPPORT_64)
  include(cmake/api_examples.cmake)
  foreach(name utf8 json numbers writer literals)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE jb::base unity)
    jb_warnings(test_${name})
    if(name STREQUAL "literals")
      set(jb_literal_fixture "${JHBASE_HAL_ROOT}/tests/fixtures/device_tree/cases.json")
      add_test(NAME literals COMMAND test_literals "${jb_literal_fixture}")
    else()
      add_test(NAME ${name} COMMAND test_${name})
    endif()
    set_tests_properties(${name} PROPERTIES TIMEOUT 60 LABELS "basic")
  endforeach()
  set(jb_probe_c "${CMAKE_CURRENT_BINARY_DIR}/probes/$<CONFIG>/literal_probe.c")
  set(jb_probe_cpp "${CMAKE_CURRENT_BINARY_DIR}/probes/$<CONFIG>/literal_probe.cpp")
  add_custom_command(OUTPUT "${jb_probe_c}" "${jb_probe_cpp}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/probes/$<CONFIG>"
    COMMAND test_literals --emit-probe "${jb_probe_c}" "${jb_literal_fixture}"
    COMMAND "${CMAKE_COMMAND}" -E copy "${jb_probe_c}" "${jb_probe_cpp}"
    DEPENDS test_literals "${jb_literal_fixture}" VERBATIM)
  # One target runs the generator first: Visual Studio copies the rule into each
  # probe project, and MSBuild would otherwise run both copies at once.
  add_custom_target(jb_literal_probes DEPENDS "${jb_probe_c}" "${jb_probe_cpp}")
  foreach(language c cpp)
    add_executable(test_literal_probe_${language} "${jb_probe_${language}}")
    add_dependencies(test_literal_probe_${language} jb_literal_probes)
    jb_warnings(test_literal_probe_${language})
    add_test(NAME literal_probe_${language} COMMAND test_literal_probe_${language})
    set_tests_properties(literal_probe_${language} PROPERTIES TIMEOUT 30 LABELS "basic")
  endforeach()
  foreach(example json_numbers json_writer)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/example-sources")
    configure_file("examples/${example}.c" "example-sources/${example}.cpp" COPYONLY)
    foreach(language c cpp)
      if(language STREQUAL "c")
        set(jb_example_source "examples/${example}.c")
      else()
        set(jb_example_source "${CMAKE_CURRENT_BINARY_DIR}/example-sources/${example}.cpp")
      endif()
      add_executable(example_${example}_${language} "${jb_example_source}")
      target_link_libraries(example_${example}_${language} PRIVATE jb::base)
      target_compile_features(example_${example}_${language} PRIVATE cxx_std_17)
      jb_warnings(example_${example}_${language})
      add_test(NAME example_${example}_${language} COMMAND example_${example}_${language})
      set_tests_properties(example_${example}_${language} PROPERTIES TIMEOUT 30 LABELS "basic")
    endforeach()
  endforeach()
  if(NOT MSVC)
    target_link_libraries(test_numbers PRIVATE m)
    add_executable(test_allocation tests/test_allocation.c)
    target_link_libraries(test_allocation PRIVATE jb::base)
    target_link_options(test_allocation PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc
      -Wl,--wrap=realloc -Wl,--wrap=free)
    jb_warnings(test_allocation)
    add_test(NAME allocation COMMAND test_allocation)
    set_tests_properties(allocation PROPERTIES TIMEOUT 30 LABELS "basic")
  endif()
  add_executable(test_cxx tests/test_cxx.cpp)
  target_link_libraries(test_cxx PRIVATE jb::base)
  target_compile_features(test_cxx PRIVATE cxx_std_17)
  set_target_properties(test_cxx PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(test_cxx PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(test_cxx PRIVATE -Wall -Wextra -Werror -Wpedantic)
  endif()
  add_test(NAME cxx COMMAND test_cxx)
  set_tests_properties(cxx PROPERTIES TIMEOUT 30 LABELS "basic")
  add_executable(test_float_format tests/test_float_format.cpp)
  target_link_libraries(test_float_format PRIVATE jb::base)
  target_compile_features(test_float_format PRIVATE cxx_std_17)
  jb_warnings(test_float_format)
  add_test(NAME float_format COMMAND test_float_format)
  set_tests_properties(float_format PROPERTIES TIMEOUT 120 LABELS "basic")
  add_test(NAME abi COMMAND "${CMAKE_COMMAND}"
    "-DJB_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
    "-DJB_WORK=${CMAKE_CURRENT_BINARY_DIR}/abi"
    "-DJB_CC=${CMAKE_C_COMPILER}" "-DJB_CXX=${CMAKE_CXX_COMPILER}"
    "-DJB_MSVC=${MSVC}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_abi.cmake")
  set_tests_properties(abi PROPERTIES TIMEOUT 60 LABELS "basic")
  if(NOT JHBASE_SANITIZERS AND NOT JHBASE_FUZZ)
    add_test(NAME install COMMAND "${CMAKE_COMMAND}"
      "-DJB_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}" "-DJB_BUILD=${CMAKE_BINARY_DIR}"
      "-DJB_WORK=${CMAKE_CURRENT_BINARY_DIR}/install-check" "-DJB_CONFIG=$<CONFIG>"
      "-DJB_GENERATOR=${CMAKE_GENERATOR}" "-DJB_PLATFORM=${CMAKE_GENERATOR_PLATFORM}"
      "-DJB_CC=${CMAKE_C_COMPILER}" "-DJB_CXX=${CMAKE_CXX_COMPILER}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_install.cmake")
    set_tests_properties(install PROPERTIES TIMEOUT 300 LABELS "basic")
  endif()
  # The VS Code workspace: task documents against the driver and the driver
  # actions on a scratch copy; nested builds skip this check themselves.
  if(JHBASE_VSCODE_CHECK AND NOT JHBASE_SANITIZERS AND NOT JHBASE_FUZZ)
    add_test(NAME vscode_workspace COMMAND "${CMAKE_COMMAND}"
      "-DJB_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
      "-DJB_SCRATCH=${CMAKE_CURRENT_BINARY_DIR}/vscode-check"
      "-DJB_TOOLS_ROOT=${JHBASE_TOOLS_ROOT}" "-DJB_UNITY_ROOT=${JHBASE_UNITY_ROOT}"
      "-DJB_HAL_ROOT=${JHBASE_HAL_ROOT}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_vscode_workspace.cmake")
    set_tests_properties(vscode_workspace PROPERTIES TIMEOUT 900 LABELS "tooling")
  endif()
  if(JHBASE_LTO)
    set_property(TARGET test_utf8 test_json test_numbers test_writer test_literals test_cxx
                 PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
endif()
